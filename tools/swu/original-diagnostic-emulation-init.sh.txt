#!/bin/sh
# Emulation-only PID1. No vendor daemons, SDIO modules or network clients.
mount -t proc proc /proc
mount -t sysfs sysfs /sys
mount -t devtmpfs devtmpfs /dev
[ -c /dev/ttyS2 ] || mknod /dev/ttyS2 c 4 66
exec </dev/ttyS2 >/dev/ttyS2 2>&1
echo RV1106_OFFLINE_USERSPACE_READY
/bin/busybox | head -n 1
echo 'Offline emulator shell; original vendor startup has not been run.'
exec /bin/sh
